function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const standingsSheet = ss.getSheetByName('Website'); const weeksSheet = ss.getSheetByName('Website_Weeks'); const playersSheet = ss.getSheetByName('Website_Players'); const historySheet = ss.getSheetByName('Website_History'); const response = { league: "REVERSE COWBOYS DraftKings NFL League", season: 2026, standings: getSheetData(standingsSheet), weeks: getSheetData(weeksSheet), players: getSheetData(playersSheet), history: getSheetData(historySheet), updated: new Date().toISOString() }; return ContentService .createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } function getSheetData(sheet) { if (!sheet) { return []; } const data = sheet.getDataRange().getValues(); if (data.length < 2) { return []; } const headers = data[0]; return data.slice(1) .filter(row => row.some(cell => cell !== "")) .map(row => { const obj = {}; headers.forEach((header, index) => { obj[header] = row[index]; }); return obj; }); }
